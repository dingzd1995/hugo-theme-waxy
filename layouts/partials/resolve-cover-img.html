{{- /* 对应原主题 functions.php 的 waxy_resolve_cover_img() + getFirstImg()，但只解析"真实" 封面图：front matter 的 img 字段，否则取正文里第一张 的图片地址。 没有真实封面图时的装饰性兜底（图池 / canvas 气泡）改到 resolve-post-bg.html 里处理， og:image 用的爬虫兜底图在 resolve-og-image.html 里处理，两边都会先调用本 partial。 入参: page（. 本身即可），返回图片 URL 字符串，没有真实封面图则返回空字符串 */ -}} {{- $img := "" -}} {{- if .Params.img -}} {{- $img = .Params.img -}} {{- else -}} {{- /* render-image.html 懒加载开启时正文里的 是 data-src="真实图" src="占位图"， 必须优先找 data-src，不然会把懒加载占位图当成封面——之前用 ]+src="..." 匹配， "data-src" 里也含有子串 "src="，贪婪匹配会先吃到那个假的，是踩过的坑 */ -}} {{- $dataSrcTags := findRE "data-src=\"[^\"]+\"" .Content 1 -}} {{- if $dataSrcTags -}} {{- $img = replaceRE "^data-src=\"([^\"]+)\"$" "$1" (index $dataSrcTags 0) -}} {{- else -}} {{- $srcTags := findRE "]+ src=\"[^\"]+\"" .Content 1 -}} {{- if $srcTags -}} {{- $img = replaceRE "^.* src=\"([^\"]+)\".*$" "$1" (index $srcTags 0) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $img -}}
